#==================================================
# CONFIGURACIÓN Y CONECTIVIDAD DEL NODO 1
#==================================================

#Mostrar dispositivos conectados por Tailscale
tailscale status

#Comprobar conexión con el Nodo 1
tailscale ping 100.120.213.18

#Comprobar conexión con el Nodo 2
tailscale ping 100.87.77.76

#Comprobar conexión con el Nodo 3
tailscale ping 100.108.44.95

#Actualizar el repositorio
git status
git pull origin main

#Crear el archivo local de credenciales
cp config/patroni/.env.example config/patroni/.env
nano config/patroni/.env


#==================================================
# CONSTRUCCIÓN DE LA IMAGEN EN EL NODO 1
#==================================================

#Construir imagen en el Nodo 1
NODE_NAME=node1 docker compose --env-file config/patroni/.env -f docker/docker-compose.node.yml build

#Comprobar versión de PostgreSQL
docker run --rm --entrypoint postgres proyecto1-patroni:pg16.15-patroni4.1.5 --version

#Comprobar versión de Patroni
docker run --rm --entrypoint patroni proyecto1-patroni:pg16.15-patroni4.1.5 --version


#==================================================
# INICIO DE ETCD EN EL NODO 1
#==================================================

#Iniciar etcd en el Nodo 1
NODE_NAME=node1 docker compose --env-file config/patroni/.env -f docker/docker-compose.node.yml up -d etcd


#==================================================
# INICIO DE PATRONI Y EXPORTERS EN EL NODO 1
#==================================================

#Iniciar servicios del Nodo 1
NODE_NAME=node1 docker compose --env-file config/patroni/.env -f docker/docker-compose.node.yml up -d --build postgres_patroni node_exporter postgres_exporter


#==================================================
# VERIFICACIÓN DEL CLÚSTER DESDE EL NODO 1
#==================================================

#Mostrar el estado de Patroni desde el Nodo 1
docker exec -it pg_node1 patronictl -c /etc/patroni.yml list

#Mostrar contenedores activos en el Nodo 1
docker ps


#==================================================
# CARGA INICIAL DE LA BASE DE DATOS
# Ejecutar solamente cuando el Nodo 1 sea Leader
#==================================================

#Cargar estructura y datos de prueba
docker exec -i pg_node1 psql -U postgres -p 5432 -d postgres < scripts/dataset.sql

#Mostrar tablas creadas
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "\dt"

#Verificar clientes cargados
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "SELECT * FROM clientes ORDER BY id;"

#Verificar productos cargados
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "SELECT id, nombre, precio, stock FROM productos ORDER BY id;"


#==================================================
# ACCESO A LA CONSOLA POSTGRESQL DEL NODO 1
#==================================================

#Entrar a psql en el Nodo 1
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres

#Salir de psql
\q


#==================================================
# COMPROBAR SI EL NODO 1 ES LÍDER O RÉPLICA
#==================================================

#Comprobar estado del Nodo 1
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "SELECT pg_is_in_recovery();"

#Resultado:
# f = Leader, permite escrituras
# t = Replica, solo permite lecturas


#==================================================
# PRUEBA DE REPLICACIÓN INICIADA EN EL NODO 1
# Ejecutar la escritura solamente si el Nodo 1 es Leader
#==================================================

#Eliminar registro de prueba desde el Nodo 1
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "DELETE FROM clientes WHERE correo = 'replicacion@prueba.local' RETURNING *;"


#==================================================
# VERIFICACIONES DE LA REPLICACIÓN INICIADA EN EL NODO 2

#==================================================

#Comprobar inserción en el Nodo 1
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "SELECT * FROM clientes WHERE correo = 'node2@prueba.local';"

#Comprobar actualización en el Nodo 1
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "SELECT * FROM clientes WHERE correo = 'node2@prueba.local';"

#Comprobar eliminación en el Nodo 1
docker exec -it pg_node1 psql -U postgres -p 5432 -d postgres -c "SELECT * FROM clientes WHERE correo = 'node2@prueba.local';"
